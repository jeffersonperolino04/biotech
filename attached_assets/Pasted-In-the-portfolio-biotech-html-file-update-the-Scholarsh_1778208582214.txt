**4. Add this JavaScript** before the closing `</script>` tag:

```js
// Marquee — duplicate cards for seamless infinite loop
['marquee-r1','marquee-r2'].forEach(id => {
  const track = document.getElementById(id);
  if (!track) return;
  Array.from(track.children).forEach(card => track.appendChild(card.cloneNode(true)));
});
// Initialize Lucide icons
lucide.createIcons();
